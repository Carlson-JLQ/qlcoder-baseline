import java
import semmle.code.java.dataflow.DataFlow
import semmle.code.java.dataflow.TaintTracking

private predicate isActionDatasetsCallable(Callable c) {
  c.getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.mgt", "ActionDatasets")
}

private predicate isConfigPathCall(MethodCall call) {
  (
    call.getMethod().hasName("generateConfigurationFilename") and
    (
      call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.webapp", "FusekiWebapp") or
      call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.mgt", "FusekiServerCtl")
    )
  )
  or
  (
    call.getMethod().hasName("existingConfigurationFile") and
    (
      call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.webapp", "FusekiWebapp") or
      call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.mgt", "FusekiServerCtl")
    )
  )
}

private predicate hasFixedValidation(Callable c) {
  exists(MethodCall call |
    call.getEnclosingCallable() = c and
    (
      (
        call.getMethod().hasName("serviceName") and
        call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.server", "Validators")
      )
      or
      call.getMethod().hasName("isValidServiceName")
      or
      (
        call.getMethod().hasName("allowConfigFiles") and
        (
          call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.webapp", "FusekiWebapp") or
          call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.mgt", "FusekiAdmin")
        )
      )
    )
  )
}

class RequestDatasetNameSource extends DataFlow::Node {
  RequestDatasetNameSource() {
    exists(MethodCall call |
      call.getMethod().hasName("getRequestParameter") and
      call.getMethod().getDeclaringType().hasQualifiedName("org.apache.jena.fuseki.servlets", "HttpAction") and
      isActionDatasetsCallable(call.getEnclosingCallable()) and
      (
        exists(StringLiteral lit |
          lit = call.getArgument(0) and
          lit.getValue() = "dbName"
        )
        or
        call.getArgument(0).(VarAccess).getVariable().hasName("paramDatasetName")
      ) and
      this.asExpr() = call
    )
  }
}

class ServiceLiteralSource extends DataFlow::Node {
  ServiceLiteralSource() {

      call.getMethod().hasName("getLexicalForm") and

      this.asExpr() = call
    )
  }
}

class Source extends DataFlow::Node {

    this instanceof ServiceLiteralSource
  }
}

class Sink extends DataFlow::Node {
  Sink() {
    exists(MethodCall call |
      isConfigPathCall(call) and

  }

module FusekiDatasetPathTraversalConfig implements DataFlow::ConfigSig {
  predicate isSource(DataFlow::Node source) { source instanceof Source }

  predicate isSink(DataFlow::Node sink) { sink instanceof Sink }

  predicate isBarrier(DataFlow::Node node) { node instanceof Sanitizer }

  predicate isAdditionalFlowStep(DataFlow::Node n1, DataFlow::Node n2) {

  }
